cmake_minimum_required(VERSION 3.21)
project(ornith_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
list(APPEND CMAKE_PREFIX_PATH /opt/rocm)
set(GPU_TARGETS "${CMAKE_HIP_ARCHITECTURES}")
find_package(hip CONFIG REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_STATIC OFF CACHE BOOL "" FORCE)
set(GGML_HIP ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_HIP_GRAPHS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA_FORCE_MMQ ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${ORNITH_LLAMA_SOURCE}/src/models/qwen35moe.cpp")
  message(FATAL_ERROR "Build the pinned dependency using ./build.sh first")
endif()
add_subdirectory("${ORNITH_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
set(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(strata_reference_kernels STATIC IMPORTED)
set_target_properties(strata_reference_kernels PROPERTIES IMPORTED_LOCATION "${STRATA_ROOT}/build-hip/libstrata_kernels.a")
add_library(strata_reference_core STATIC IMPORTED)
set_target_properties(strata_reference_core PROPERTIES IMPORTED_LOCATION "${STRATA_ROOT}/build-hip/libstrata_core.a")
add_library(strata_reference_qwen35_gpu STATIC IMPORTED)
set_target_properties(strata_reference_qwen35_gpu PROPERTIES IMPORTED_LOCATION "${STRATA_ROOT}/build-hip/libstrata_qwen35_gpu.a")
enable_testing()
add_executable(ornith_primitives primitives.cpp "${STRATA_ROOT}/src/core/hip_budget.cpp")
target_include_directories(ornith_primitives BEFORE PRIVATE "${STRATA_ROOT}/include/strata/hip_compat" "${STRATA_ROOT}/include")
target_compile_definitions(ornith_primitives PRIVATE STRATA_USE_HIP=1)
target_link_libraries(ornith_primitives PRIVATE strata_reference_qwen35_gpu strata_reference_kernels strata_reference_core ggml ggml-cpu ggml-hip hip::host ${CMAKE_DL_LIBS})
target_link_options(ornith_primitives PRIVATE -Wl,--export-dynamic)
add_test(NAME ornith_primitives COMMAND ornith_primitives)
add_executable(ornith_reference reference.cpp ../../src/core/hip_budget.cpp)
target_include_directories(ornith_reference PRIVATE "${ORNITH_LLAMA_SOURCE}/src" ../../include)
target_link_libraries(ornith_reference PRIVATE llama hip::host ${CMAKE_DL_LIBS})
target_link_options(ornith_reference PRIVATE -Wl,--export-dynamic)
